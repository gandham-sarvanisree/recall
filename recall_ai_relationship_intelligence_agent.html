Technical implementation reference for Next.js 15 App Router, PostgreSQL pgvector Schema, and Recall Hindsight Engine.

-- Production PostgreSQL Database Schema with pgvector
CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE IF NOT EXISTS contacts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL,
  name TEXT NOT NULL,
  email TEXT UNIQUE,
  company TEXT,
  role TEXT,
  health_score INT DEFAULT 85,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS recall_memories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  bank_id TEXT NOT NULL,
  memory_type TEXT CHECK (memory_type IN ('episodic', 'semantic', 'procedural')),
  content TEXT NOT NULL,
  embedding vector(1536),
  is_commitment BOOLEAN DEFAULT FALSE,
  due_date DATE,
  confidence FLOAT DEFAULT 0.95,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_memories_bank ON recall_memories(bank_id);
